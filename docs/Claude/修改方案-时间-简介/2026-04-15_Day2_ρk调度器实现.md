```cpp
struct RiskSchedulerParams {
    // sigmoid 参数
    double gamma = 5.0;
    double rho_0 = 0.3;
    // 变化率限制（每周期最大变化量，占满量程比例）
    double rate_limit_per_step = 0.05;
    // 调度范围
    double Q_eta_min = 0.0;
    double Q_eta_max = 10.0;
    double eta_bar_max = 0.05;     // 与 slosh_height_max 一致
    double delta_eta_bar = 0.02;
    double beta = 0.3;             // 速度折扣上限
    // 风险权重
    double w_h = 0.4;   // 液面预测风险
    double w_e = 0.3;   // 模态能量风险
    double w_t = 0.3;   // 终点接近度
    double w_r = 0.7;   // r_k 权重
    double w_u = 0.3;   // u_k 权重
    // fallback 触发阈值
    double u_threshold_high = 0.8;
    int u_high_count_trigger = 10;
    double imu_timeout_s = 0.1;
    double a_uncert_max = 0.5;     // u_k 归一化分母 (m/s²)
    // fallback 固定参数
    double Q_eta_fix = 5.0;
    double eta_bar_fix = 0.04;
};

struct RiskSchedulerOutput {
    double Q_eta_k;
    double eta_bar_k;
    double v_ref_eff_k;
    double rho_k;
    double r_k;
    double u_k;
    double s_rho;
    bool fallback_active;
};

class RiskScheduler {
public:
    void init(const RiskSchedulerParams& params);

    // outer loop 每周期调用一次
    // 输入：上一周期冻结的 monitor 值 + 当前 IMU 观测 + 运动学
    RiskSchedulerOutput update(
        double h_pred_max_prev,   // 上一周期 /slosh/height_pred_max
        double E_slosh_prev,      // 上一周期模态能量 (eta_x^2 + eta_y^2)
        double d_goal,            // 当前到目标距离
        double a_y_imu,           // IMU ay（已扣零偏）
        double v_omega,           // 运动学 v*omega
        ros::Time imu_stamp,      // 最新 IMU 时间戳（用于超时检测）
        bool imu_bias_ready,      // /slosh/imu_ay_bias_ready
        double v_ref              // 当前参考速度
    );

private:
    double sigmoid(double x);
    double clip(double val, double lo, double hi);
    RiskSchedulerParams params_;
    RiskSchedulerOutput prev_output_;
    int u_high_count_ = 0;
    bool initialized_ = false;
};
```

## Step 2：实现 RiskScheduler::update()

```cpp
RiskSchedulerOutput RiskScheduler::update(...) {
    RiskSchedulerOutput out;

    // --- Fallback 检测 ---
    bool imu_timeout = (ros::Time::now() - imu_stamp).toSec() > params_.imu_timeout_s;
    bool bias_not_ready = !imu_bias_ready;
    double u_k_raw = clip(std::fabs(a_y_imu - v_omega) / params_.a_uncert_max, 0.0, 1.0);
    if (u_k_raw > params_.u_threshold_high) u_high_count_++;
    else u_high_count_ = 0;
    bool u_persistent = (u_high_count_ >= params_.u_high_count_trigger);

    out.fallback_active = imu_timeout || bias_not_ready || u_persistent;

    if (out.fallback_active) {
        out.Q_eta_k     = params_.Q_eta_fix;
        out.eta_bar_k   = params_.eta_bar_fix;
        out.v_ref_eff_k = v_ref;
        out.rho_k = out.r_k = out.u_k = out.s_rho = 0.0;
        prev_output_ = out;
        return out;
    }

    // --- r_k：physical risk（来自冻结 monitor rollout）---
    double h_max = params_.eta_bar_max;
    double E_max = h_max * h_max * 2.0;  // 两轴满量程估计
    double d_goal_thresh = 1.0;          // m，与 terminal_recovery.enter_distance 对齐

    double h_risk = clip(h_pred_max_prev / h_max, 0.0, 1.0);
    double e_risk = clip(E_slosh_prev / E_max, 0.0, 1.0);
    double t_risk = clip((d_goal_thresh - d_goal) / d_goal_thresh, 0.0, 1.0);
    out.r_k = params_.w_h * h_risk + params_.w_e * e_risk + params_.w_t * t_risk;
    out.r_k = clip(out.r_k, 0.0, 1.0);

    // --- u_k：excitation uncertainty（独立 IMU 观测）---
    out.u_k = u_k_raw;

    // --- ρ_k ---
    out.rho_k = clip(params_.w_r * out.r_k + params_.w_u * out.u_k, 0.0, 1.0);
    out.s_rho = sigmoid(params_.gamma * (out.rho_k - params_.rho_0));

    // --- 期望值 ---
    double Q_des   = params_.Q_eta_min + (params_.Q_eta_max - params_.Q_eta_min) * out.s_rho;
    double bar_des = params_.eta_bar_max - params_.delta_eta_bar * out.s_rho;
    double vref_des = v_ref * (1.0 - params_.beta * out.s_rho);

    // --- 变化率限制 ---
    double dQ   = (params_.Q_eta_max - params_.Q_eta_min) * params_.rate_limit_per_step;
    double dbar = params_.delta_eta_bar * params_.rate_limit_per_step;
    double dvref = v_ref * params_.beta * params_.rate_limit_per_step;

    out.Q_eta_k   = clip(Q_des,   prev_output_.Q_eta_k   - dQ,   prev_output_.Q_eta_k   + dQ);
    out.eta_bar_k = clip(bar_des, prev_output_.eta_bar_k - dbar, prev_output_.eta_bar_k + dbar);
    out.v_ref_eff_k = clip(vref_des, prev_output_.v_ref_eff_k - dvref, prev_output_.v_ref_eff_k + dvref);

    prev_output_ = out;
    return out;
}
```

## Step 3：接入 LocalPlannerROS

**文件**：`src/local_planner_ros.cpp`

改动点（只改这几处，不动 QP 主体）：

```cpp
// 1. 头部新增
#include "scout_local_planner/risk_scheduler.h"

// 2. 类成员新增
RiskScheduler risk_scheduler_;
RiskSchedulerOutput risk_output_;

// 3. init() 里从 param server 读参数并初始化
RiskSchedulerParams rs_params;
nh_.param("risk_scheduler/gamma", rs_params.gamma, 5.0);
// ... 其他参数同理
risk_scheduler_.init(rs_params);

// 4. controlLoop() 里，在 MPCSolver::solve() 之前调用 outer loop
risk_output_ = risk_scheduler_.update(
    slosh_state_.h_pred_max_prev,   // 上一周期保存的值
    slosh_state_.E_slosh_prev,
    dist_to_goal_,
    slosh_estimator_.getAyImu(),    // 已有接口
    current_state_.v * current_state_.omega,
    last_imu_stamp_,
    imu_ay_bias_ready_,
    ref_speed_
);

// 5. 把 risk_output_ 注入 MPC 参数
mpc_solver_.setSloshWeight(risk_output_.Q_eta_k);
mpc_solver_.setEtaBar(risk_output_.eta_bar_k);
path_handler_.setEffectiveRefSpeed(risk_output_.v_ref_eff_k);

// 6. controlLoop() 末尾保存本周期 monitor 供下周期使用
slosh_state_.h_pred_max_prev = slosh_state_.h_pred_max;   // 已有话题对应值
slosh_state_.E_slosh_prev    = slosh_state_.eta_x * slosh_state_.eta_x
                             + slosh_state_.eta_y * slosh_state_.eta_y;
```
